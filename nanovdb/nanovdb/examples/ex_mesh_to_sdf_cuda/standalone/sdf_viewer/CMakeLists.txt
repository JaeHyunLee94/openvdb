cmake_minimum_required(VERSION 3.24)
# CUDA 12 can JIT this PTX on newer GPUs, including Blackwell.
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 90-virtual)
endif()
project(mesh_sdf_viewer LANGUAGES CXX CUDA)
get_filename_component(OPENVDB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
set(NANOVDB_INCLUDE_DIR "${OPENVDB_ROOT}/nanovdb" CACHE PATH "Directory containing nanovdb/NanoVDB.h")
find_package(ZLIB REQUIRED)
add_library(sdf_renderer SHARED renderer.cu)
target_include_directories(sdf_renderer PRIVATE "${NANOVDB_INCLUDE_DIR}")
target_compile_features(sdf_renderer PRIVATE cxx_std_17 cuda_std_17)
target_compile_definitions(sdf_renderer PRIVATE NANOVDB_USE_ZIP)
target_compile_options(sdf_renderer PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda;--expt-relaxed-constexpr>)
target_link_libraries(sdf_renderer PRIVATE ZLIB::ZLIB)
find_path(BLOSC_INCLUDE_DIR blosc.h)
find_library(BLOSC_LIBRARY blosc)
if(BLOSC_INCLUDE_DIR AND BLOSC_LIBRARY)
  target_include_directories(sdf_renderer PRIVATE "${BLOSC_INCLUDE_DIR}")
  target_link_libraries(sdf_renderer PRIVATE "${BLOSC_LIBRARY}")
  target_compile_definitions(sdf_renderer PRIVATE NANOVDB_USE_BLOSC)
endif()
# Optional converter: compiles the existing driver in this independent build.
option(BUILD_MESH_CONVERTER "Build the existing MeshToSDF driver without editing it" OFF)
if(BUILD_MESH_CONVERTER)
  set(CCCL_SOURCE_DIR "${OPENVDB_ROOT}/build/release/_deps/cccl-src" CACHE PATH "CCCL source tree used by MeshToSDF")
  add_executable(sdf_from_mesh ../../mesh_to_sdf_cuda.cpp ../../mesh_to_sdf_cuda_kernels.cu)
  target_include_directories(sdf_from_mesh PRIVATE "${NANOVDB_INCLUDE_DIR}")
  target_compile_features(sdf_from_mesh PRIVATE cxx_std_17 cuda_std_17)
  if(EXISTS "${CCCL_SOURCE_DIR}/libcudacxx/include/cuda")
    target_include_directories(sdf_from_mesh BEFORE PRIVATE "${CCCL_SOURCE_DIR}/thrust" "${CCCL_SOURCE_DIR}/cub" "${CCCL_SOURCE_DIR}/libcudacxx/include")
    target_compile_definitions(sdf_from_mesh PRIVATE THRUST_HOST_SYSTEM=THRUST_HOST_SYSTEM_CPP)
  endif()
  target_compile_options(sdf_from_mesh PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda;--expt-relaxed-constexpr>)
endif()
option(BUILD_VIEWER_TESTS "Build analytic test fixtures" OFF)
if(BUILD_VIEWER_TESTS)
  add_executable(sdf_test_grids make_test_grids.cpp)
  target_include_directories(sdf_test_grids PRIVATE "${NANOVDB_INCLUDE_DIR}")
  target_compile_features(sdf_test_grids PRIVATE cxx_std_17)
endif()
