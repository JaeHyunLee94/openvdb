cmake_minimum_required(VERSION 3.24)
# CUDA 12 can JIT this PTX on newer GPUs, including Blackwell.
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 90-virtual)
endif()
project(mesh_sdf_viewer LANGUAGES CXX CUDA)
get_filename_component(OPENVDB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
set(NANOVDB_INCLUDE_DIR "${OPENVDB_ROOT}/nanovdb" CACHE PATH "Directory containing nanovdb/NanoVDB.h")
find_package(ZLIB REQUIRED)
add_library(sdf_renderer SHARED renderer.cu)
target_include_directories(sdf_renderer PRIVATE "${NANOVDB_INCLUDE_DIR}")
target_compile_features(sdf_renderer PRIVATE cxx_std_17 cuda_std_17)
target_compile_definitions(sdf_renderer PRIVATE NANOVDB_USE_ZIP)
target_compile_options(sdf_renderer PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda;--expt-relaxed-constexpr>)
target_link_libraries(sdf_renderer PRIVATE ZLIB::ZLIB)
find_path(BLOSC_INCLUDE_DIR blosc.h)
find_library(BLOSC_LIBRARY blosc)
if(BLOSC_INCLUDE_DIR AND BLOSC_LIBRARY)
  target_include_directories(sdf_renderer PRIVATE "${BLOSC_INCLUDE_DIR}")
  target_link_libraries(sdf_renderer PRIVATE "${BLOSC_LIBRARY}")
  target_compile_definitions(sdf_renderer PRIVATE NANOVDB_USE_BLOSC)
endif()
# Optional converter: compiles the existing driver in this independent build.
option(BUILD_MESH_CONVERTER "Build the existing MeshToSDF driver without editing it" OFF)
if(BUILD_MESH_CONVERTER)
  set(CCCL_SOURCE_DIR "${OPENVDB_ROOT}/build/release/_deps/cccl-src" CACHE PATH "CCCL source tree used by MeshToSDF")
  add_executable(sdf_from_mesh ../../mesh_to_sdf_cuda.cpp ../../mesh_to_sdf_cuda_kernels.cu)
  target_include_directories(sdf_from_mesh PRIVATE "${NANOVDB_INCLUDE_DIR}")
  target_compile_features(sdf_from_mesh PRIVATE cxx_std_17 cuda_std_17)
  if(EXISTS "${CCCL_SOURCE_DIR}/libcudacxx/include/cuda")
    target_include_directories(sdf_from_mesh BEFORE PRIVATE "${CCCL_SOURCE_DIR}/thrust" "${CCCL_SOURCE_DIR}/cub" "${CCCL_SOURCE_DIR}/libcudacxx/include")
    target_compile_definitions(sdf_from_mesh PRIVATE THRUST_HOST_SYSTEM=THRUST_HOST_SYSTEM_CPP)
  endif()
  target_compile_options(sdf_from_mesh PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda;--expt-relaxed-constexpr>)
endif()
# GNU/Linux-only opt-in allocation adapter for real-mesh tests exceeding VRAM.
option(BUILD_MANAGED_MESH_CONVERTER "Build a Unified Memory MeshToSDF validation driver" OFF)
if(BUILD_MANAGED_MESH_CONVERTER)
  if(NOT BUILD_MESH_CONVERTER OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Managed converter requires BUILD_MESH_CONVERTER and GNU/Linux")
  endif()
  add_executable(sdf_from_mesh_managed ../../mesh_to_sdf_cuda.cpp ../../mesh_to_sdf_cuda_kernels.cu managed_alloc.cu)
  get_target_property(CONVERTER_INCLUDES sdf_from_mesh INCLUDE_DIRECTORIES)
  get_target_property(CONVERTER_DEFINITIONS sdf_from_mesh COMPILE_DEFINITIONS)
  target_include_directories(sdf_from_mesh_managed PRIVATE ${CONVERTER_INCLUDES})
  if(CONVERTER_DEFINITIONS)
    target_compile_definitions(sdf_from_mesh_managed PRIVATE ${CONVERTER_DEFINITIONS})
  endif()
  target_compile_features(sdf_from_mesh_managed PRIVATE cxx_std_17 cuda_std_17)
  target_compile_options(sdf_from_mesh_managed PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda;--expt-relaxed-constexpr>)
  target_link_options(sdf_from_mesh_managed PRIVATE -Wl,--wrap=cudaMalloc -Wl,--wrap=cudaMallocAsync -Wl,--wrap=cudaFree -Wl,--wrap=cudaFreeAsync)
endif()
option(BUILD_VIEWER_TESTS "Build analytic test fixtures" OFF)
if(BUILD_VIEWER_TESTS)
  add_executable(sdf_test_grids make_test_grids.cpp)
  target_include_directories(sdf_test_grids PRIVATE "${NANOVDB_INCLUDE_DIR}")
  target_compile_features(sdf_test_grids PRIVATE cxx_std_17)
endif()
option(BUILD_ROOT_FILL_VALIDATION "Test actual PropagateSign root filling against independent truth" OFF)
if(BUILD_ROOT_FILL_VALIDATION)
  enable_testing()
  add_executable(sdf_validate_root_fill validate_root_fill.cu)
  add_executable(sdf_analyze_mesh_root analyze_mesh_root.cpp)
  target_include_directories(sdf_analyze_mesh_root PRIVATE "${NANOVDB_INCLUDE_DIR}")
  target_compile_features(sdf_analyze_mesh_root PRIVATE cxx_std_17)
  add_executable(sdf_check_cube_mesh_roots check_cube_mesh_roots.cpp)
  target_include_directories(sdf_check_cube_mesh_roots PRIVATE "${NANOVDB_INCLUDE_DIR}")
  target_compile_features(sdf_check_cube_mesh_roots PRIVATE cxx_std_17)
  add_test(NAME sdf_root_fill COMMAND sdf_validate_root_fill)
  target_include_directories(sdf_validate_root_fill PRIVATE "${NANOVDB_INCLUDE_DIR}")
  target_compile_features(sdf_validate_root_fill PRIVATE cxx_std_17 cuda_std_17)
  target_compile_options(sdf_validate_root_fill PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda;--expt-relaxed-constexpr>)
endif()
